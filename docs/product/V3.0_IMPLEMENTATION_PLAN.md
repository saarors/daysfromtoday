**步骤 5：创建 Worker 代码**
```typescript
// workers/short-link-resolver/src/index.ts
export interface Env {
  SHORT_LINKS: KVNamespace;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    
    // 提取短代码（/c/abc123 → abc123）
    const shortCode = url.pathname.replace('/c/', '');
    
    if (!shortCode) {
      return new Response('Invalid short code', { status: 400 });
    }
    
    // 从 KV 读取卡片数据
    const cardData = await env.SHORT_LINKS.get(shortCode);
    
    if (!cardData) {
      return new Response('Short link not found', { status: 404 });
    }
    
    // 解析数据
    const data = JSON.parse(cardData);
    
    // 增加浏览次数
    data.view_count = (data.view_count || 0) + 1;
    await env.SHORT_LINKS.put(shortCode, JSON.stringify(data));
    
    // 重定向到完整页面
    return Response.redirect(
      `https://www.daysfromtoday.ai/share/${shortCode}`,
      302
    );
  },
};
```

```typescript
// lib/feature-flags.ts
import { get } from '@vercel/edge-config';

export async function getFeatureFlags() {
  const rolloutPercentage = await get<number>('rollout_percentage') || 0;
  const enabledFeatures = await get<string[]>('enabled_features') || [];
  
  return {
    rolloutPercentage,
    enabledFeatures,
  };
}

export async function isFeatureEnabled(featureName: string): Promise<boolean> {
  const flags = await getFeatureFlags();
  return flags.enabledFeatures.includes(featureName);
}
```

```typescript
// lib/ratelimit.ts
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,

// 匿名用户限流：10 次/小时
export const anonymousRatelimit = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(10, '1 h'),
  analytics: true,
  prefix: 'ratelimit:anonymous',
});

// 登录用户限流：50 次/小时
export const authenticatedRatelimit = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(50, '1 h'),

  prefix: 'ratelimit:authenticated',
});

// 短链接生成限流：5 次/天
export const shortLinkRatelimit = new Ratelimit({
  redis,
  limiter: Ratelimit.slidingWindow(5, '24 h'),

  prefix: 'ratelimit:shortlink',
});
```

```typescript
import { NextRequest } from 'next/server';

  try {
    const { searchParams } = new URL(request.url);
```

```typescript
        const newCard: CardData = {
          ...card,
          id: nanoid(),
          createdAt: new Date().toISOString(),
          updatedAt: new Date().toISOString(),
        };
        
        set((state) => ({
          cards: [...state.cards, newCard]
        }));
        
        return newCard;
      },
      
      updateCard: (id, updates) => {
        set((state) => ({
          cards: state.cards.map(c => 
            c.id === id 
              ? { ...c, ...updates, updatedAt: new Date().toISOString() } 
              : c
          )
        }));
      },
      
      deleteCard: (id) => {
        set((state) => ({
          cards: state.cards.filter(c => c.id !== id)
        }));
      },
      
      getCard: (id) => {
        return get().cards.find(c => c.id === id);
      },
      
      getAllCards: () => {
        return get().cards;
      },
      
      clearCards: () => {
        set({ cards: [] });
      },
    }),
    {
      name: 'goal-cards-storage',
      storage: createJSONStorage(() => {
        // SSR 安全的 storage
        if (typeof window === 'undefined') {
          return {
            getItem: () => null,
            setItem: () => {},
            removeItem: () => {}
          };
        }
        return localStorage;
      }),
```

```typescript
export async function middleware(request: NextRequest) {
  const host = request.headers.get('host') || '';

  // 防止 Vercel 默认域名被索引
  if (host.endsWith('.vercel.app')) {

  if (supabaseResponse) {
```

```typescript
 * 同步本地卡片到云端

export async function syncLocalToCloud(userId: string): Promise<{ synced: number; errors: number }> {
  // 1. 读取本地数据
  const localCards = useGoalCards.getState().getAllCards();
  
  if (localCards.length === 0) {
    return { synced: 0, errors: 0 };
  }
  
  // 2. 上传到 Supabase
  const supabase = createClient();
  let synced = 0;
  let errors = 0;
  
  for (const card of localCards) {
    try {
      const { error } = await supabase
        .from('goal_cards')
        .insert({
          user_id: userId,
          template_id: card.templateId,
          title: card.content.title,
          target_date: card.content.targetDate,
          days_count: card.content.daysCount,
          goal_text: card.content.goalText,
          start_date: card.content.startDate,
          emoji: card.content.emoji,
          custom_background_url: card.customBackground?.imageUrl,
          custom_background_id: card.customBackground?.imageId,
        });
      
      if (error) {
        console.error('Sync error:', error);
        errors++;
      } else {
        synced++;
      }
    } catch (error) {
      console.error('Sync error:', error);
      errors++;
    }
  }
  
  // 3. 清空本地缓存（可选，根据策略决定）
  // useGoalCards.getState().clearCards();
  
  return { synced, errors };
}
```

```typescript
// components/CustomBackgroundUploader.tsx
'use client';

import { useState } from 'react';
import Image from 'next/image';

interface CustomBackgroundUploaderProps {
  onUpload: (imageUrl: string, imageId: string) => void;
}

export function CustomBackgroundUploader({ onUpload }: CustomBackgroundUploaderProps) {
  const [uploading, setUploading] = useState(false);
  const [preview, setPreview] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  
  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    
    setError(null);
    
    // 1. 验证文件类型
    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) {
      setError('仅支持 JPG、PNG、WebP 格式');
      return;
    }
    
    // 2. 验证文件大小
    if (file.size > 5 * 1024 * 1024) {
      setError('图片大小不能超过 5MB');
      return;
    }
    
    // 3. 预览
    const reader = new FileReader();
    reader.onload = (e) => setPreview(e.target?.result as string);
    reader.readAsDataURL(file);
    
    // 4. 上传
    setUploading(true);
    try {
      const formData = new FormData();
      formData.append('file', file);
      
      const response = await fetch('/api/upload/background', {
        method: 'POST',
        body: formData,
      });
      
      if (!response.ok) {
        throw new Error('Upload failed');
      }
      
      const { imageUrl, imageId } = await response.json();
      onUpload(imageUrl, imageId);
    } catch (error) {
      setError('上传失败，请重试');
    } finally {
      setUploading(false);
    }
  };
  
  return (
    <div className="space-y-4">
      <label className="block cursor-pointer">
        <div className="relative aspect-video bg-gray-100 rounded-lg overflow-hidden border-2 border-dashed border-gray-300 hover:border-blue-500 transition-colors">
          {preview ? (
            <Image
              src={preview}
              alt="Preview"
              fill
              className="object-cover"
            />
          ) : (
            <div className="absolute inset-0 flex flex-col items-center justify-center">
              <svg className="w-12 h-12 text-gray-400 mb-2" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 16l4.586-4.586a2 2 0 012.828 0L16 16m-2-2l1.586-1.586a2 2 0 012.828 0L20 14m-6-6h.01M6 20h12a2 2 0 002-2V6a2 2 0 00-2-2H6a2 2 0 00-2 2v12a2 2 0 002 2z" />
              </svg>
              <p className="text-gray-600 font-medium">点击上传背景图片</p>
              <p className="text-gray-400 text-sm mt-1">支持 JPG、PNG、WebP，最大 5MB</p>
            </div>
          )}
          
          {uploading && (
            <div className="absolute inset-0 bg-black/50 flex items-center justify-center">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-white"></div>
            </div>
          )}
        </div>
        
        <input
          type="file"
          accept="image/jpeg,image/png,image/webp"
          className="hidden"
          onChange={handleFileSelect}
          disabled={uploading}
        />
      </label>
      
      {error && (
        <p className="text-red-500 text-sm">⚠️ {error}</p>
      )}
    </div>
  );
}
```

#### Day 3-4：图片压缩 + 上传到 R2

```typescript
// app/api/upload/background/route.ts
import { NextRequest, NextResponse } from 'next/server';
import sharp from 'sharp';
import { nanoid } from 'nanoid';
import { createClient } from '@/lib/supabase/server';
import { uploadMultiFormatImage } from '@/lib/r2-client';

export async function POST(request: NextRequest) {
  try {
    // 1. 验证用户认证
    const supabase = createClient();
    const { data: { user } } = await supabase.auth.getUser();
    
    if (!user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    
    // 2. 读取文件
    const formData = await request.formData();
    const file = formData.get('file') as File;
    
    if (!file) {
      return NextResponse.json({ error: 'No file uploaded' }, { status: 400 });
    }
    
    // 3. 压缩图片
    const buffer = await file.arrayBuffer();
    const compressedBuffer = await sharp(Buffer.from(buffer))
      .resize(1920, 1080, {
        fit: 'inside',
        withoutEnlargement: true,
      })
      .webp({ quality: 85 })
      .toBuffer();
    
    // 4. 上传到 R2
    const imageId = nanoid();
    const r2KeyBase = `backgrounds/${user.id}/${imageId}`;
    const result = await uploadMultiFormatImage(compressedBuffer, r2KeyBase);
    
    return NextResponse.json({
      imageUrl: result.webp.url,
      imageId,
    });
  } catch (error) {
    console.error('Upload error:', error);
    return NextResponse.json(
      { error: 'Upload failed' },
      { status: 500 }
    );
  }
}
```
