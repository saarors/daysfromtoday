### **4. profiles 表（用户配额管理）**

```sql
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  
  -- 配额管理
  goal_cards_limit INTEGER DEFAULT 3, -- Free: 3, Pro: 无限
  goal_cards_count INTEGER DEFAULT 0,
  subscription_tier VARCHAR(20) DEFAULT 'free', -- free/pro/enterprise
  
  -- 用户偏好
  preferred_language VARCHAR(10) DEFAULT 'zh',
  preferred_assistant VARCHAR(20) DEFAULT 'twinkle',
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- RPC 函数：增加配额
CREATE OR REPLACE FUNCTION increment_goal_cards_count(user_id UUID)
RETURNS void AS $$
BEGIN
  UPDATE public.profiles
  SET goal_cards_count = goal_cards_count + 1
  WHERE id = user_id;
END;
$$ LANGUAGE plpgsql;

-- RPC 函数：减少配额
CREATE OR REPLACE FUNCTION decrement_goal_cards_count(user_id UUID)
RETURNS void AS $$
BEGIN
  UPDATE public.profiles
  SET goal_cards_count = GREATEST(0, goal_cards_count - 1)
  WHERE id = user_id;
END;
$$ LANGUAGE plpgsql;

-- RPC 函数：检查配额
CREATE OR REPLACE FUNCTION check_goal_cards_quota(user_id UUID)
RETURNS BOOLEAN AS $$
DECLARE
  current_count INTEGER;
  max_limit INTEGER;
BEGIN
  SELECT goal_cards_count, goal_cards_limit
  INTO current_count, max_limit
  FROM public.profiles
  WHERE id = user_id;
  
  RETURN current_count < max_limit;
END;
$$ LANGUAGE plpgsql;
```

### **路由访问控制**

```typescript
// middleware.ts

const publicPaths = [
  '/login',
  '/register',
  '/api/auth',
  '/days',           // 日期计算页面（所有语言）
  '/calculator',     // 日期计算器
  '/blog',           // 博客
  '/holidays',       // 节假日
  '/',               // 首页（故事发现可匿名访问）
];

const protectedPaths = [
  '/wishlist',       // 愿望清单 - 必须登录
  '/api/ai',         // AI API - 必须登录
];

export async function middleware(request: NextRequest) {
  const pathname = request.nextUrl.pathname;
  
  // 检查是否为公开路径
  const isPublic = publicPaths.some(path => 
    pathname.includes(path) || pathname === '/' || pathname.match(/^\/(zh|en)$/)
  );
  
  if (isPublic) {
    return intlMiddleware(request);
  }
  
  // 检查是否为受保护路径
  const isProtected = protectedPaths.some(path => pathname.includes(path));
  
  if (isProtected) {
    const { data: { session } } = await supabase.auth.getSession();
    
    if (!session) {
      const locale = pathname.split('/')[1] || 'zh';
      return NextResponse.redirect(new URL(`/${locale}/login`, request.url));
    }
  }
  
  return intlMiddleware(request);
}
```

## 🔧 提示词匹配逻辑

```typescript
// lib/ai-prompt-matcher.ts

interface PromptMatchParams {
  assistantType: 'twinkle' | 'labubu' | 'jobs';
  goalText: string;
  days: number;
  targetDate: string;
}

/**
 * 检测输入语言
 */
function detectLanguage(text: string): 'zh' | 'en' {
  const chineseChars = text.match(/[\u4e00-\u9fa5]/g);
  return chineseChars && chineseChars.length > 3 ? 'zh' : 'en';
}

/**
 * 计算目标阶段
 */
function calculateStage(days: number): 'start' | 'in_progress' | 'near_end' {
  // Phase 3 简化版：全部返回 start
  return 'start';
}

/**
 * 获取匹配的提示词
 */
export async function getMatchedPrompt(params: PromptMatchParams) {
  const { assistantType, goalText, days, targetDate } = params;
  
  const language = detectLanguage(goalText);
  const stage = calculateStage(days);
  
  // 从数据库查询提示词
  const { data, error } = await supabase
    .from('ai_prompt_templates')
    .select('*')
    .eq('assistant_type', assistantType)
    .eq('goal_type', 'general')
    .eq('stage', stage)
    .eq('output_language', language)
    .eq('is_active', true)
    .order('version', { ascending: false })
    .limit(1)
    .single();
  
  if (error || !data) {
    throw new Error('Prompt not found');
  }
  
  // 替换模板变量
  const userPrompt = data.user_prompt_template
    .replace('{goalText}', goalText)
    .replace('{days}', days.toString())
    .replace('{targetDate}', targetDate);
  
  return {
    systemPrompt: data.system_prompt,
    userPrompt,
    language,
    stage,
  };
}
```

### **DeepSeek API 定价**

- 输入 Token: ~200 (prompt + 用户输入)
